using System.Collections.Generic;
using UnityEngine;

public class InventoryManager : MonoBehaviour
{
    public List<ItemData> inventory = new List<ItemData>();
    public int maxSlots = 10;

    public void AddItem(ItemData item)
    {
        for (int i = 0; i < inventory.Count; i++)
        {
            if (inventory[i].itemType == item.itemType && inventory[i].stackSize < item.stackMax)
            {
                inventory[i].stackSize += item.stackSize;
                return;
            }
        }

        if (inventory.Count < maxSlots)
        {
            inventory.Add(new ItemData(item));
        }
    }

    public void UseItem(ItemData item)
    {
        for (int i = 0; i < inventory.Count; i++)
        {
            if (inventory[i].itemType == item.itemType && inventory[i].stackSize > 0)
            {
                inventory[i].stackSize -= 1;
                if (inventory[i].stackSize <= 0)
                {
                    inventory.RemoveAt(i);
                }
                return;
            }
        }
    }

    public bool HasItem(ItemData item)
    {
        foreach (var invItem in inventory)
        {
            if (invItem.itemType == item.itemType && invItem.stackSize > 0)
            {
                return true;
            }
        }
        return false;
    }
}

[System.Serializable]
public class ItemData
{
    public string itemName;
    public Sprite icon;
    public int stackSize = 1;
    public int stackMax = 99;
    public string description;
    public ItemType itemType;

    public ItemData(ItemData item)
    {
        itemName = item.itemName;
        icon = item.icon;
        stackSize = item.stackSize;
        stackMax = item.stackMax;
        description = item.description;
        itemType = item.itemType;
    }
}

using UnityEngine;

public class InventorySlot : MonoBehaviour
{
    public ItemData slotItem;
    public int slotIndex;

    private InventoryManager inventoryManager;

    void Start()
    {
        inventoryManager = FindObjectOfType<InventoryManager>();
    }

    public void OnMouseDown()
    {
        if (slotItem != null)
        {
            inventoryManager.UseItem(slotItem);
            UpdateSlot();
        }
    }

    public void UpdateSlot()
    {
        if (inventoryManager.inventory.Count > slotIndex && inventoryManager.inventory[slotIndex].stackSize > 0)
        {
            slotItem = inventoryManager.inventory[slotIndex];
            GetComponentInChildren<TextMeshProUGUI>().text = slotItem.stackSize.ToString();
            GetComponentInChildren<Image>().sprite = slotItem.icon;
        }
        else
        {
            slotItem = null;
            GetComponentInChildren<TextMeshProUGUI>().text = "";
            GetComponentInChildren<Image>().sprite = null;
        }
    }

    public void AddItem(ItemData item)
    {
        if (slotItem == null || slotItem.itemType == item.itemType && slotItem.stackSize < item.stackMax)
        {
            slotItem = new ItemData(item);
            UpdateSlot();
        }
    }
}

using System.Collections.Generic;
using UnityEngine;

public class InventoryManager : MonoBehaviour
{
    public List<ItemData> inventory = new List<ItemData>();
    public int maxSlots = 10;

    public void AddItem(ItemData item)
    {
        for (int i = 0; i < inventory.Count; i++)
        {
            if (inventory[i].itemType == item.itemType && inventory[i].stackSize < item.stackMax)
            {
                inventory[i].stackSize += item.stackSize;
                return;
            }
        }

        if (inventory.Count < maxSlots)
        {
            inventory.Add(new ItemData(item));
        }
    }

    public void UseItem(ItemData item)
    {
        for (int i = 0; i < inventory.Count; i++)
        {
            if (inventory[i].itemType == item.itemType && inventory[i].stackSize > 0)
            {
                inventory[i].stackSize -= 1;
                if (inventory[i].stackSize <= 0)
                {
                    inventory.RemoveAt(i);
                }
                return;
            }
        }
    }

    public bool HasItem(ItemData item)
    {
        foreach (var invItem in inventory)
        {
            if (invItem.itemType == item.itemType && invItem.stackSize > 0)
            {
                return true;
            }
        }
        return false;
    }
}

        description = item.description;

using UnityEngine;
using UnityEngine.UI;

public class InventoryUI : MonoBehaviour
{
    public Transform inventorySlotsParent;
    public GameObject inventorySlotPrefab;
    private InventoryManager inventoryManager;

    void Start()
    {
        inventoryManager = FindObjectOfType<InventoryManager>();
        if (inventoryManager == null)
        {
            Debug.LogError("InventoryManager not found. Please assign it in the Inspector.");
            return;
        }
        UpdateInventoryUI();
    }

    public void UpdateInventoryUI()
    {
        foreach (Transform child in inventorySlotsParent)
        {
            Destroy(child.gameObject);
        }

        for (int i = 0; i < inventoryManager.maxSlots; i++)
        {
            GameObject slotObject = Instantiate(inventorySlotPrefab, inventorySlotsParent);
            InventorySlot slot = slotObject.GetComponent<InventorySlot>();
            if (slot == null)
            {
                Debug.LogError("InventorySlot component not found on prefab. Please check your prefab.");
                return;
            }
            slot.slotIndex = i;
            slot.UpdateSlot();
        }
    }
}
